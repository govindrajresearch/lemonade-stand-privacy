/**
 * Lemonade Stand — MBA skills reports for groups (classes and competitions).
 * Reads groups, members and decision logs from Firestore and writes PDFs to Google Drive:
 *   Lemonade Stand Archive / Reports / <College> / <Group (CODE)> / <Student (Player ID)>.pdf
 *   … plus "Group summary" in each group folder, "College summary" in each college folder,
 *   and the spreadsheet "Lemonade Stand reports" (Students and Groups tabs) with a link to every PDF.
 * Run reports() any time (it only redoes groups with new data), or reportsAll() to redo everything.
 * setup() in Code.gs also runs it every Monday morning. Uses api(), val(), BASE and FOLDER from Code.gs.
 * Covers decisions still in Firebase (the last 180 days); older days are in the monthly archive sheets.
 */
var REPORTS_FOLDER = 'Reports';
var CUR_INFO = { USD: [1, '$'], EUR: [1, '€'], GBP: [1, '£'], CAD: [1, 'C$'], AUD: [2, 'A$'], CHF: [2, 'CHF '], INR: [50, '₹'], CNY: [5, 'CN¥'],
  JPY: [100, '¥'], KRW: [1000, '₩'], IDR: [10000, 'Rp '], BRL: [5, 'R$'], MXN: [20, 'MX$'], NGN: [1000, '₦'] };

function reportsAll() { reports(true); }

function reports(force) {
  var started = Date.now(), props = PropertiesService.getScriptProperties();
  var groups = query('groups', null), byCollege = {}, rows = [], groupRows = [], cache = {}, done = 0, left = 0;
  groups.forEach(function (gd) {
    var code = gd.name.split('/').pop(), f = gd.fields || {};
    var g = { code: code, name: val(f.name), college: val(f.college) || 'No college', batch: val(f.batch), ownerName: val(f.ownerName) };
    var members = query('members', { field: 'code', op: 'EQUAL', value: { stringValue: code } }).map(function (d) {
      var mf = d.fields || {}; return { pid: val(mf.pid), name: val(mf.name), college: val(mf.college), t: new Date(val(mf.t)) };
    });
    if (!members.length) return;
    var students = members.map(function (mb) { return studentReport(mb, cache); });
    var stamp = students.map(function (s) { return s.pid + ':' + s.days.length; }).join('|');
    var dirty = force || props.getProperty('rep_' + code) !== stamp;
    if (dirty && Date.now() - started > 4.5 * 60 * 1000) { left++; dirty = false; }   // out of time: the next run continues
    var folder = path([REPORTS_FOLDER, g.college, g.name + ' (' + code + ')']);
    students.sort(function (a, b) { return (b.sk ? b.sk.overall : -1) - (a.sk ? a.sk.overall : -1); });
    students.forEach(function (s, i) {
      s.rank = s.sk ? i + 1 : null;
      s.url = dirty ? savePdf(folder, s.name + ' (' + s.pid + ')', studentHtml(s, g, students)) : fileUrl(folder, s.name + ' (' + s.pid + ')');
      rows.push([new Date(), g.college, g.name, code, g.batch, s.name, s.pid, s.days.length].concat(s.sk ? [s.sk.overall, s.sk.grade]
        .concat(['fin', 'ops', 'mkt', 'risk', 'eth', 'learn'].map(function (id) { var a = s.sk.areas.filter(function (x) { return x.id === id; })[0]; return a ? a.score : ''; })) : ['', 'Under 5 days', '', '', '', '', '', '']).concat([s.url]));
    });
    var gUrl = dirty ? savePdf(folder, 'Group summary', groupHtml(g, students)) : fileUrl(folder, 'Group summary');
    if (dirty) { props.setProperty('rep_' + code, stamp); done++; }
    var scored = students.filter(function (s) { return s.sk; });
    groupRows.push([g.college, g.name, code, g.batch, g.ownerName, students.length, scored.length ? Math.round(avgOf(scored, function (s) { return s.sk.overall; })) : '', gUrl]);
    (byCollege[g.college] = byCollege[g.college] || []).push({ g: g, students: students });
  });
  Object.keys(byCollege).forEach(function (c) { savePdf(path([REPORTS_FOLDER, c]), 'College summary', collegeHtml(c, byCollege[c])); });
  writeIndex(rows, groupRows);
  Logger.log('Groups: ' + groups.length + ', redone: ' + done + (left ? ', still to do (run again): ' + left : ''));
}

/* ---------- Firestore ---------- */
function query(collection, filter) {
  var q = { from: [{ collectionId: collection }] };
  if (filter) q.where = { fieldFilter: { field: { fieldPath: filter.field }, op: filter.op, value: filter.value } };
  return api('post', BASE + ':runQuery', { structuredQuery: q }).filter(function (x) { return x.document; }).map(function (x) { return x.document; });
}
/* All decision days a member saved since joining the group (research records are batches of days). */
function studentReport(mb, cache) {
  if (!cache[mb.pid]) cache[mb.pid] = query('research', { field: 'pid', op: 'EQUAL', value: { stringValue: mb.pid } })
    .map(function (d) { var f = d.fields || {}, days = []; try { days = JSON.parse(val(f.days) || '[]'); } catch (e) {} return { t: new Date(val(f.t)), days: days }; })
    .sort(function (a, b) { return a.t - b.t; });
  var days = [];
  cache[mb.pid].forEach(function (r) { if (r.t >= mb.t) days = days.concat(r.days); });
  var k = SK.kpis(days);
  return { pid: mb.pid, name: mb.name || mb.pid, college: mb.college, days: days, k: k, sk: SK.skills(k), cur: days.length ? days[days.length - 1].cur || 'INR' : 'INR' };
}

/* ---------- Drive ---------- */
function path(parts) {
  var it = DriveApp.getFoldersByName(FOLDER), f = it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER);
  parts.forEach(function (p) { p = String(p).replace(/[\/\\]/g, '-').slice(0, 100); var s = f.getFoldersByName(p); f = s.hasNext() ? s.next() : f.createFolder(p); });
  return f;
}
function savePdf(folder, name, html) {
  var blob = Utilities.newBlob(html, MimeType.HTML, name + '.html').getAs(MimeType.PDF).setName(name + '.pdf');
  var old = folder.getFilesByName(name + '.pdf');
  if (old.hasNext()) {   // replace the contents in place, so links already opened or shared show the latest report
    var f = old.next();
    UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files/' + f.getId() + '?uploadType=media', { method: 'patch', contentType: 'application/pdf',
      payload: blob.getBytes(), headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() } });
    while (old.hasNext()) old.next().setTrashed(true);
    return f.getUrl();
  }
  return folder.createFile(blob).getUrl();
}
function fileUrl(folder, name) { var f = folder.getFilesByName(name + '.pdf'); return f.hasNext() ? f.next().getUrl() : ''; }
function writeIndex(rows, groupRows) {
  var root = path([REPORTS_FOLDER]), files = root.getFilesByName('Lemonade Stand reports'), ss;
  if (files.hasNext()) ss = SpreadsheetApp.open(files.next());
  else { ss = SpreadsheetApp.create('Lemonade Stand reports'); DriveApp.getFileById(ss.getId()).moveTo(root); ss.getSheets()[0].setName('Students'); ss.insertSheet('Groups'); }
  var put = function (name, head, data) {
    var sh = ss.getSheetByName(name); sh.clear(); sh.appendRow(head); sh.setFrozenRows(1); sh.getRange(1, 1, 1, head.length).setFontWeight('bold');
    if (data.length) sh.getRange(2, 1, data.length, head.length).setValues(data);
  };
  put('Students', ['Updated', 'College', 'Group', 'Code', 'Batch', 'Student', 'Player ID', 'Days', 'Overall /100', 'Grade', 'Finance', 'Operations', 'Marketing', 'Risk', 'Ethics', 'Learning', 'Report PDF'], rows);
  put('Groups', ['College', 'Group', 'Code', 'Batch', 'Created by', 'Students', 'Average overall', 'Group summary PDF'], groupRows);
}

var CSS = '<style>body{font-family:Arial,sans-serif;font-size:10pt;color:#1f2937}h1{font-size:17pt;color:#9a6700;margin:0}h2{font-size:12pt;margin:16px 0 6px;color:#374151}' +
  'table{border-collapse:collapse;width:100%}td,th{border:1px solid #d1d5db;padding:5px 6px;text-align:left;vertical-align:top}th{background:#fef3c7}' +
  '.big{font-size:22pt;font-weight:bold;color:#15803d}.muted{color:#6b7280;font-size:8.5pt}.n{text-align:right}.box{border:1px solid #9ca3af;height:90px}</style>';
function e(s) { return String(s == null ? '' : s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
function money(gc, cur) {
  var c = CUR_INFO[cur] || CUR_INFO.INR, d = c[0] <= 5 ? 2 : 0;   // cents for dollar-like currencies
  return gc == null ? '—' : (gc < 0 ? '−' : '') + c[1] + (Math.abs(gc) / 100 * c[0]).toLocaleString('en-IN', { minimumFractionDigits: d, maximumFractionDigits: d });
}
function pc(x) { return x == null ? '—' : Math.round(x * 100) + '%'; }
function stars(s) { var n = Math.round(s / 20); return '★★★★★'.slice(0, n) + '☆☆☆☆☆'.slice(n); }
function avgOf(a, f) { return a.length ? a.reduce(function (s, x) { return s + f(x); }, 0) / a.length : 0; }
function head(title, sub) { return '<html><head><meta charset="utf-8">' + CSS + '</head><body><h1>🍋 ' + e(title) + '</h1><div class="muted">' + sub + '</div>'; }
var FOOT = '<p class="muted">Lemonade Stand: Business Sim · reports by Dr. Govindraj R. Mane, Department of MBA, KLS Gogte Institute of Technology, Belagavi. ' +
  'Scores are computed from the student\'s own game decisions; use them as a starting point for feedback, not as a final grade.</p></body></html>';
var AREAS = [['fin', 'Finance'], ['ops', 'Operations'], ['mkt', 'Marketing'], ['risk', 'Risk'], ['eth', 'Ethics'], ['learn', 'Learning']];
function areaScore(s, id) { if (!s.sk) return ''; var a = s.sk.areas.filter(function (x) { return x.id === id; })[0]; return a ? a.score : '—'; }

  var h = head('MBA Skills Report', e(g.name) + ' (' + g.code + ')' + (g.batch ? ' · ' + e(g.batch) : '') + ' · ' + e(g.college) + ' · generated ' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'd MMM yyyy'));

    '<tr><th>Rank in group</th><td>' + (s.rank ? s.rank + ' of ' + all.filter(function (x) { return x.sk; }).length : '—') + '</td><th>College</th><td>' + e(s.college) + '</td></tr></table>';

    r('Break-even', k.breakEven == null ? '—' : k.breakEven === 0 ? 'No fixed costs yet' : k.breakEven + ' cups a day', 'Average sold: ' + Math.round(k.avgCups || 0) + ' cups') +
    r('Conversion rate', pc(k.conversion), 'Passers-by who bought') +
    r('Price elasticity', k.elasticity == null ? 'Not measured' : k.elasticity.toFixed(2), 'Below −1: price-sensitive customers') +
    r('Lost sales', pc(k.lostSales), 'Buyers turned away (stock-out or capacity)') +
    r('Waste and spoilage', pc(k.waste), 'Ingredient cost thrown away') +
    r('Stock-out days · over-made days', pc(k.stockOutDays) + ' · ' + pc(k.overMadeDays), 'Matching supply to demand') +
    r('Average taste', Math.round(k.taste || 0), 'Product quality (0–100)') +

}

function groupHtml(g, students) {
  var sc = students.filter(function (s) { return s.sk; });
  var h = head('Group Summary', e(g.name) + ' (' + g.code + ')' + (g.batch ? ' · ' + e(g.batch) : '') + ' · ' + e(g.college) + ' · created by ' + e(g.ownerName) + ' · ' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'd MMM yyyy'));
  h += '<p>' + students.length + ' students joined · ' + sc.length + ' with at least 5 days of play.</p>';
  if (sc.length) {
    var avgs = AREAS.map(function (a) { var w = sc.filter(function (s) { return areaScore(s, a[0]) !== '—'; }); return [a[1], w.length ? Math.round(avgOf(w, function (s) { return areaScore(s, a[0]); })) : null, a[0]]; }).filter(function (x) { return x[1] != null; });
    var weak = avgs.slice().sort(function (a, b) { return a[1] - b[1]; })[0];
    h += '<h2>Class averages</h2><table><tr><th>Overall</th>' + avgs.map(function (a) { return '<th>' + a[0] + '</th>'; }).join('') + '</tr><tr><td class="n"><b>' + Math.round(avgOf(sc, function (s) { return s.sk.overall; })) + '</b></td>' +
      avgs.map(function (a) { return '<td class="n">' + a[1] + '</td>'; }).join('') + '</tr></table>' +
      '<p><b>Weakest area for the class: ' + weak[0] + '</b> (average ' + weak[1] + '). A good topic for the next class discussion.</p>' +
      '<h2>Top in each area</h2><table>' + avgs.map(function (a) { var best = sc.slice().sort(function (x, y) { return (areaScore(y, a[2]) || 0) - (areaScore(x, a[2]) || 0); })[0]; return '<tr><td>' + a[0] + '</td><td>' + e(best.name) + '</td><td class="n">' + areaScore(best, a[2]) + '</td></tr>'; }).join('') + '</table>';
  }
  var byC = {};
  sc.forEach(function (st) { conceptScores(st.days, st.k, st.cur).list.forEach(function (c) { (byC[c.name] = byC[c.name] || { area: c.area, v: [] }).v.push(c.s); }); });
  var names = Object.keys(byC);
  if (names.length) {
    names.sort(function (a, b) { return avgOf(byC[a].v, function (x) { return x; }) - avgOf(byC[b].v, function (x) { return x; }); });
    h += '<h2>Class average by concept (weakest first)</h2><table><tr><th>Area</th><th>Concept</th><th>Students</th><th>Average</th></tr>' + names.map(function (n) {
      var a = Math.round(avgOf(byC[n].v, function (x) { return x; })); return '<tr><td class="muted">' + e(byC[n].area) + '</td><td>' + e(n) + '</td><td class="n">' + byC[n].v.length + '</td><td class="n"><b>' + a + '</b> ' + stars(a) + '</td></tr>'; }).join('') + '</table>' +
      '<p>Suggested topics for class discussion: <b>' + names.slice(0, 3).map(e).join('</b>, <b>') + '</b>.</p>';
  }
  h += '<h2>Ranking</h2><table><tr><th>#</th><th>Student</th><th>Player ID</th><th>Days</th><th>Overall</th>' + AREAS.map(function (a) { return '<th>' + a[1] + '</th>'; }).join('') + '<th>Grade</th></tr>' +
    students.map(function (s) { return '<tr><td>' + (s.rank || '—') + '</td><td>' + e(s.name) + '</td><td>' + s.pid + '</td><td class="n">' + s.days.length + '</td><td class="n"><b>' + (s.sk ? s.sk.overall : '—') + '</b></td>' +
      AREAS.map(function (a) { return '<td class="n">' + areaScore(s, a[0]) + '</td>'; }).join('') + '<td>' + (s.sk ? s.sk.grade : 'Under 5 days') + '</td></tr>'; }).join('') + '</table>';
  return h + FOOT;
}

function collegeHtml(college, list) {
  var h = head('College Summary', e(college) + ' · ' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'd MMM yyyy')), everyone = [];
  h += '<h2>Groups</h2><table><tr><th>Group</th><th>Code</th><th>Batch</th><th>Students</th><th>Average overall</th>' + AREAS.map(function (a) { return '<th>' + a[1] + '</th>'; }).join('') + '</tr>' +
    list.map(function (x) {
      var sc = x.students.filter(function (s) { return s.sk; }); everyone = everyone.concat(sc.map(function (s) { return { s: s, g: x.g }; }));
      return '<tr><td>' + e(x.g.name) + '</td><td>' + x.g.code + '</td><td>' + e(x.g.batch) + '</td><td class="n">' + x.students.length + '</td><td class="n"><b>' + (sc.length ? Math.round(avgOf(sc, function (s) { return s.sk.overall; })) : '—') + '</b></td>' +
        AREAS.map(function (a) { var w = sc.filter(function (s) { return typeof areaScore(s, a[0]) === 'number'; }); return '<td class="n">' + (w.length ? Math.round(avgOf(w, function (s) { return areaScore(s, a[0]); })) : '—') + '</td>'; }).join('') + '</tr>';
    }).join('') + '</table>';
  everyone.sort(function (a, b) { return b.s.sk.overall - a.s.sk.overall; });
  h += '<h2>Top 10 students</h2><table><tr><th>#</th><th>Student</th><th>Group</th><th>Overall</th><th>Grade</th></tr>' +
    everyone.slice(0, 10).map(function (x, i) { return '<tr><td>' + (i + 1) + '</td><td>' + e(x.s.name) + '</td><td>' + e(x.g.name) + '</td><td class="n">' + x.s.sk.overall + '</td><td>' + x.s.sk.grade + '</td></tr>'; }).join('') + '</table>';
  return h + FOOT;
}

/* ---------- Concept-by-concept scores: each MBA concept, what it is, how the game uses it, and what the student did ----------
 * conceptScores(days, k, cur) → [{ area, name, what, applied, s (0–100), good, bad, tip }] for concepts seen in play,
 * plus .missing: names of concepts the student never practised. Scores are rules of thumb for feedback, not exam marks. */
var WX = { hot: [1.20, 4], sunny: [1.00, 3], cloudy: [0.95, 2], cool: [0.85, 1], rainy: [0.85, 1], stormy: [0.80, 1] };   // [fair-price factor, ideal ice]
function conceptScores(days, k, cur) {
  var O = days.filter(function (d) { return !d.cl; }), out = [], missing = [];
  var cl = function (x) { return Math.max(0, Math.min(1, x)); }, P = function (x) { return Math.round(x * 100) + '%'; }, M = function (x) { return money(x, cur); };
  var sum = function (a, f) { return a.reduce(function (s, x) { return s + f(x); }, 0); }, avg = function (a, f) { return a.length ? sum(a, f) / a.length : 0; };
  var share = function (a, f) { return a.length ? a.filter(f).length / a.length : 0; };
  var corr = function (pts) {
    if (pts.length < 3) return null;
    var mx = avg(pts, function (p) { return p[0]; }), my = avg(pts, function (p) { return p[1]; });
    var sxy = sum(pts, function (p) { return (p[0] - mx) * (p[1] - my); }), sxx = sum(pts, function (p) { return (p[0] - mx) * (p[0] - mx); }), syy = sum(pts, function (p) { return (p[1] - my) * (p[1] - my); });
    return sxx > 0 && syy > 0 ? sxy / Math.sqrt(sxx * syy) : null;
  };
  var fair = function (d) { return 100 * (WX[d.f] || WX[d.w] || [1])[0]; };
  function add(area, name, what, applied, r) {
    if (!r) { missing.push(name); return; }
    r.s = Math.round(100 * cl(r.s)); r.area = area; r.name = name; r.what = what; r.applied = applied; out.push(r);
  }
  if (!O.length) return { list: [], missing: [] };
  var rev = sum(O, function (d) { return d.rev; }), last = days[days.length - 1], first = days[0];

/* ---------- Scores: copied from the app (app/src/main/assets/www/js/journal.js). tests/run.js checks they match. ---------- */
var SK = (function () {
  function sum(a, f) { return a.reduce(function (s, x) { return s + f(x); }, 0); }
  function avg(a, f) { return a.length ? sum(a, f) / a.length : 0; }
  /* Least-squares slope of y on x. */
  function slope(pts) {
    if (pts.length < 2) return null;
    var mx = avg(pts, function (p) { return p[0]; }), my = avg(pts, function (p) { return p[1]; });
    var sxx = sum(pts, function (p) { return (p[0] - mx) * (p[0] - mx); });
    if (sxx < 1e-9) return null;
    return sum(pts, function (p) { return (p[0] - mx) * (p[1] - my); }) / sxx;
  }

  function kpis(days) {
    var open = days.filter(function (d) { return !d.cl; });
    var k = { days: days.length, open: open.length };
    if (!open.length) return k;
    k.profit = sum(days, function (d) { return d.np; });
    k.avgProfit = k.profit / days.length;
    k.profitableShare = days.filter(function (d) { return d.np > 0; }).length / days.length;
    var rev = sum(open, function (d) { return d.rev; }), cogs = sum(open, function (d) { return d.cogs; }), sv = sum(open, function (d) { return d.sv; });
    k.grossMargin = rev > 0 ? (rev - cogs) / rev : null;
    k.avgPrice = sv > 0 ? rev / sv : null;
    k.costPerCup = sv > 0 ? cogs / sv : null;
    k.contribution = k.avgPrice != null ? k.avgPrice - k.costPerCup : null;
    var fixed = avg(open, function (d) { return d.fee + d.ad; });
    k.breakEven = k.contribution > 0 ? Math.ceil(fixed / k.contribution) : null;
    k.avgCups = sv / open.length;
    var pb = sum(open, function (d) { return d.pb; }), by = sum(open, function (d) { return d.by; }), ta = sum(open, function (d) { return d.ta; });
    k.conversion = pb > 0 ? by / pb : null;
    k.lostSales = by > 0 ? ta / by : null;
    var wst = sum(open, function (d) { return d.wst + d.sp; });
    k.waste = cogs + wst > 0 ? wst / (cogs + wst) : null;
    k.stockOutDays = open.filter(function (d) { return d.ta > 0; }).length / open.length;
    k.overMadeDays = open.filter(function (d) { return d.wp > 0; }).length / open.length;
    // price elasticity: slope of ln(conversion) on ln(price), using days with buyers
    var pts = open.filter(function (d) { return d.pb > 0 && d.by > 0 && d.pr > 0; }).map(function (d) { return [Math.log(d.pr), Math.log(d.by / d.pb)]; });
    var prices = {}; open.forEach(function (d) { prices[d.pr] = 1; });
    k.elasticity = pts.length >= 5 && Object.keys(prices).length >= 2 ? slope(pts) : null;
    // marketing return: extra profit on advertising days per unit of ad spend
    var adD = open.filter(function (d) { return d.ad > 0; }), noAd = open.filter(function (d) { return d.ad === 0; });
    k.adDays = adD.length;
    k.adReturn = adD.length >= 2 && noAd.length >= 2 ? (avg(adD, function (d) { return d.np + d.ad; }) - avg(noAd, function (d) { return d.np; })) / avg(adD, function (d) { return d.ad; }) : null;
    // risk and behaviour
    k.loanDays = days.filter(function (d) { return d.loan > 0; }).length / days.length;
    var afterLoss = 0, cutAfterLoss = 0;
    for (var i = 1; i < days.length; i++) if (days[i - 1].np < 0 && days[i].g === days[i - 1].g && !days[i].cl) { afterLoss++; if (days[i].pr < days[i - 1].pr) cutAfterLoss++; }
    k.cutAfterLoss = afterLoss >= 3 ? cutAfterLoss / afterLoss : null;
    var storm = open.filter(function (d) { return d.w === 'stormy' || d.w === 'rainy'; });
    k.badWeatherProfit = storm.length ? avg(storm, function (d) { return d.np; }) : null;
    // learning curve: first 7 vs latest 7 days
    if (days.length >= 14) { k.first7 = avg(days.slice(0, 7), function (d) { return d.np; }); k.last7 = avg(days.slice(-7), function (d) { return d.np; }); }
    k.taste = avg(open, function (d) { return d.tas; });
    // compliance: the officer's visits and how they were handled
    k.visits = days.filter(function (d) { return d.of; }).length;
    k.bribes = days.filter(function (d) { return d.of === 'bribe'; }).length;
    k.permit = days.some(function (d) { return d.pm; });
    k.interest = sum(days, function (d) { return d.in || 0; });
    return k;
  }

  /* ---------- MBA skill scores (0–100) with one feedback line per area ----------
   * Copied into backend/archive/Reports.gs for the teacher's PDF reports: keep the two in step. */
  var MIN_SKILL_DAYS = 5;
  function skills(k) {
    if (!k.open || k.days < MIN_SKILL_DAYS) return null;
    var cl = function (x) { return Math.max(0, Math.min(1, x)); }, P = function (x) { return Math.round(x * 100) + '%'; };
    var areas = [];
    var gm = k.grossMargin == null ? 0 : k.grossMargin;
    areas.push({ id: 'fin', name: 'Finance and accounting', concepts: 'Profitability, gross margin, working capital, leverage',
      score: 50 * k.profitableShare + 25 * (1 - k.loanDays) + 25 * cl(gm / 0.6),
      tip: k.loanDays > 0.3 ? 'Relied on loans on ' + P(k.loanDays) + ' of days. Plan cash for tomorrow\'s stock before spending (working capital).'
        : k.profitableShare < 0.6 ? 'Only ' + P(k.profitableShare) + ' of days were profitable. Compare price with cost per cup every day (contribution margin).'
        : gm < 0.4 ? 'Gross margin is thin (' + P(gm) + '). Cut ingredient cost or raise price where customers allow it.'
        : 'Strong: profitable on ' + P(k.profitableShare) + ' of days with a ' + P(gm) + ' gross margin and little borrowing.' });
    var waste = k.waste == null ? 0 : k.waste;
    areas.push({ id: 'ops', name: 'Operations and supply chain', concepts: 'Inventory, perishables, capacity, matching supply to demand',
      score: 50 * (1 - cl(waste / 0.3)) + 25 * (1 - k.stockOutDays) + 25 * (1 - k.overMadeDays),
      tip: waste > 0.15 ? P(waste) + ' of ingredient cost was wasted. Buy closer to expected demand (inventory control of perishables).'
        : k.stockOutDays > 0.3 ? 'Ran out of stock on ' + P(k.stockOutDays) + ' of days and lost sales. Keep a little safety stock on busy days.'
        : k.overMadeDays > 0.5 ? 'Poured lemonade away on ' + P(k.overMadeDays) + ' of days. Match production to the forecast.'
        : 'Strong: low waste (' + P(waste) + ') and few stock-outs. Supply matched demand well.' });
    var conv = k.conversion == null ? 0 : k.conversion;
    areas.push({ id: 'mkt', name: 'Marketing and pricing', concepts: 'Price elasticity, product quality, conversion, marketing return',
      score: 40 * cl(conv / 0.5) + 30 * k.taste / 100 + 30 * (k.adReturn == null ? 0.5 : cl(k.adReturn)),
      tip: k.taste < 75 ? 'Average taste was ' + Math.round(k.taste) + '. Balance lemons and sugar and match ice to the weather (product quality).'
        : conv < 0.3 ? 'Only ' + P(conv) + ' of passers-by bought. The price may be above what customers expect (price elasticity).'
        : k.adReturn != null && k.adReturn < 1 ? 'Advertising did not pay for itself. Measure the return before spending more (marketing ROI).'
        : !k.adDays ? 'Never tried advertising. Test it on busy days and compare the results.'
        : 'Strong: good taste, ' + P(conv) + ' of passers-by bought, and advertising paid back.' });
    var bw = k.badWeatherProfit, cut = k.cutAfterLoss;
    areas.push({ id: 'risk', name: 'Risk and decision-making', concepts: 'Forecasting, risk management, loss aversion, test and learn',
      score: 40 * (bw == null ? 0.5 : bw >= 0 ? 1 : 0.3) + 30 * (cut == null ? 0.5 : cl(1 - Math.abs(cut - 0.4) / 0.6)) + 30 * (k.elasticity != null ? 1 : 0.3),
      tip: bw != null && bw < 0 ? 'Lost money on rainy and stormy days. Scale down or close when the forecast is bad (risk management).'
        : cut != null && cut > 0.7 ? 'Cut the price after ' + P(cut) + ' of loss days. Find the cause before reacting (loss aversion).'
        : k.elasticity == null ? 'Tried too few prices to learn how customers respond. Experiment with price (test and learn).'
        : 'Strong: handled bad weather and reacted to losses calmly.' });
    if (k.visits || k.permit) {
      var bs = k.visits ? k.bribes / k.visits : 0;
      areas.push({ id: 'eth', name: 'Ethics and compliance', concepts: 'Regulatory compliance, corruption as a recurring cost, governance',
        score: 100 - 80 * bs - (!k.permit && k.visits >= 3 ? 20 : 0),
        tip: k.bribes ? 'Paid bribes on ' + k.bribes + ' of ' + k.visits + ' officer visits. A bribe comes back every second day; the permit is a one-time cost (compliance as investment).'
          : !k.permit && k.visits >= 3 ? 'Paid fines or closed on ' + k.visits + ' officer visits. A Street Vending Permit would have stopped them.'
          : 'Strong: stayed within the rules' + (k.permit ? ' and bought the Street Vending Permit.' : '.') });
    }
    if (k.first7 != null) {
      var gain = (k.last7 - k.first7) / Math.max(Math.abs(k.first7), 100);
      areas.push({ id: 'learn', name: 'Learning and growth', concepts: 'Learning curve, continuous improvement',
        score: 50 + 50 * Math.max(-1, Math.min(1, gain)),
        tip: gain > 0.1 ? 'Daily profit rose from the first week to the latest. Clear learning.' : gain < -0.1 ? 'Daily profit fell from the first week to the latest. Review what changed.' : 'Daily profit stayed about the same. Try one new idea at a time.' });
    }
    var tot = 0;
    areas.forEach(function (a) { a.score = Math.round(Math.max(0, Math.min(100, a.score))); tot += a.score; });
    var overall = Math.round(tot / areas.length);
    return { overall: overall, grade: overall >= 80 ? 'Excellent' : overall >= 65 ? 'Good' : overall >= 50 ? 'Fair' : 'Needs work', areas: areas };
  }

  return { kpis: kpis, skills: skills };
})();
