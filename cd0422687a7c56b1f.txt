/**
 * Lemonade Stand — nightly archive of class/research decision logs.
 * Moves Firestore "research" records older than 180 days into Google Sheets in your Drive
 * (folder "Lemonade Stand Archive", one spreadsheet per month), then deletes them from Firestore.
 * If the estimated Firestore storage passes 700 MB it archives more (records older than 30 days).
 * Setup: see SETUP.md. Run setup() once; it installs the nightly trigger.
 */
var PROJECT = 'lemonade-stand-356e5';
var KEEP_DAYS = 180;              // the window kept in Firebase
var FAST_KEEP_DAYS = 30;          // used when storage is getting full
var FAST_FROM_BYTES = 700 * 1024 * 1024;
var MAX_DELETES_PER_RUN = 15000;  // free plan allows 20,000 deletes a day
var FOLDER = 'Lemonade Stand Archive';
var BASE = 'https://firestore.googleapis.com/v1/projects/' + PROJECT + '/databases/(default)/documents';
var COLS = ['at', 'g', 'dif', 'cur', 'd', 'w', 'f', 'tp', 'ev', 'cl', 'pr', 'pd', 'le', 'su', 'ic', 'ad', 'pb', 'by', 'sv', 'ta', 'pi', 'wp',
  'rev', 'cogs', 'wst', 'sp', 'fee', 'lf', 'np', 'cash', 'nw', 'pur', 'tas', 'rep', 'rd', 'tier', 'loan', 'rp', 'up',
  'bk', 'rj', 'in', 'wg', 'tx', 'iv', 'fi', 'bb', 'pn', 'sz', 'of', 'hp', 'pm'];
var HEAD = ['Player ID', 'Class code', 'Group', 'Age band', 'Record', 'Saved at (UTC)'].concat(['Date and time', 'Game', 'Difficulty', 'Currency', 'Day', 'Weather', 'Forecast', 'Temperature °C',
  'Event', 'Closed', 'Price (game cents)', 'Product', 'Lemons per pitcher', 'Sugar per pitcher', 'Ice per cup', 'Advertising (game cents)', 'Passers-by', 'Buyers',
  'Cups sold', 'Turned away', 'Pitchers', 'Cups poured away', 'Revenue (gc)', 'Cost of cups sold (gc)', 'Waste (gc)', 'Spoilage (gc)', 'Stand fee (gc)',
  'Loan fees (gc)', 'Net profit (gc)', 'Cash at close (gc)', 'Net worth (gc)', 'Supplies bought (gc)', 'Taste', 'Reputation', 'Reputation change',
  'Stand tier', 'All loans owed (gc)', 'Family loan repaid (gc)', 'Upgrades owned', 'Bank loan owed (gc)', 'Moneylender owed (gc)', 'Interest (gc)',
  'Helper wage (gc)', 'Income tax (gc)', 'Investor share (gc)', 'Fine (gc)', 'Bribe (gc)', 'Closure loss (gc)', 'Assets seized (gc)', 'Officer choice', 'Helper', 'Permit']);

function setup() {
  ScriptApp.getProjectTriggers().forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('nightly').timeBased().everyDays(1).atHour(2).create();      // around 2 am in the script's time zone
  ScriptApp.newTrigger('weeklySummary').timeBased().onWeekDay(ScriptApp.WeekDay.MONDAY).atHour(8).create();
  Logger.log('Triggers installed. Records in Firebase now: ' + countRecords());
}

function api(method, url, body) {
  var r = UrlFetchApp.fetch(url, { method: method, contentType: 'application/json', muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, payload: body ? JSON.stringify(body) : undefined });
  var code = r.getResponseCode(), text = r.getContentText();
  if (code >= 300) throw new Error('Firestore ' + code + ': ' + text.slice(0, 300));
  return text ? JSON.parse(text) : {};
}

function countRecords() {
  var r = api('post', BASE + ':runAggregationQuery', { structuredAggregationQuery: { structuredQuery: { from: [{ collectionId: 'research' }] },
    aggregations: [{ alias: 'n', count: {} }] } });
  return Number(((r[0] || {}).result || {}).aggregateFields ? r[0].result.aggregateFields.n.integerValue : 0);
}

function older(cutoff, limit) {
  var r = api('post', BASE + ':runQuery', { structuredQuery: { from: [{ collectionId: 'research' }],
    where: { fieldFilter: { field: { fieldPath: 't' }, op: 'LESS_THAN', value: { timestampValue: cutoff.toISOString() } } },
    orderBy: [{ field: { fieldPath: 't' } }], limit: limit } });
  return r.filter(function (x) { return x.document; }).map(function (x) { return x.document; });
}

function val(v) {
  if (!v) return '';
  if ('stringValue' in v) return v.stringValue;
  if ('integerValue' in v) return Number(v.integerValue);
  if ('doubleValue' in v) return v.doubleValue;
  if ('timestampValue' in v) return v.timestampValue;
  return '';
}

function sheetFor(month, cache) {
  if (cache[month]) return cache[month];
  var folders = DriveApp.getFoldersByName(FOLDER), folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER);
  var name = 'Lemonade Stand decisions ' + month, files = folder.getFilesByName(name), ss;
  if (files.hasNext()) ss = SpreadsheetApp.open(files.next());
  else {
    ss = SpreadsheetApp.create(name);
    DriveApp.getFileById(ss.getId()).moveTo(folder);
    ss.getSheets()[0].setName('Days').appendRow(HEAD);
    ss.getSheets()[0].setFrozenRows(1);
  }
  return (cache[month] = ss.getSheetByName('Days'));
}

/** Runs every night. */
function nightly() {
  var started = Date.now(), n = countRecords(), sample = older(new Date(), 20);
  var avgBytes = sample.length ? sample.reduce(function (s, d) { return s + JSON.stringify(d).length; }, 0) / sample.length : 0;
  var estBytes = n * avgBytes, keep = estBytes > FAST_FROM_BYTES ? FAST_KEEP_DAYS : KEEP_DAYS;
  var cutoff = new Date(Date.now() - keep * 864e5), done = 0, rows = 0, cache = {};
  while (done < MAX_DELETES_PER_RUN && Date.now() - started < 4.5 * 60 * 1000) {
    var docs = older(cutoff, 200);
    if (!docs.length) break;
    var byMonth = {};
    docs.forEach(function (doc) {
      var f = doc.fields || {}, saved = val(f.t), month = String(saved).slice(0, 7) || 'unknown', days = [];
      try { days = JSON.parse(val(f.days) || '[]'); } catch (e) {}
      (byMonth[month] = byMonth[month] || []);
      days.forEach(function (d) { byMonth[month].push([val(f.pid), val(f.cls), val(f.grp) || 'adult', val(f.ab), doc.name.split('/').pop(), saved].concat(COLS.map(function (c) { return d[c] == null ? '' : d[c]; }))); });
    });
    Object.keys(byMonth).forEach(function (month) {        // write to Sheets first …
      var list = byMonth[month]; if (!list.length) return;
      var sh = sheetFor(month, cache);
      sh.getRange(sh.getLastRow() + 1, 1, list.length, HEAD.length).setValues(list);
      rows += list.length;
    });
    SpreadsheetApp.flush();
    api('post', BASE + ':commit', { writes: docs.map(function (d) { return { delete: d.name }; }) });   // … then delete
    done += docs.length;
  }
  var log = PropertiesService.getScriptProperties();
  log.setProperty('lastRun', JSON.stringify({ at: new Date().toISOString(), records: n, estMB: Math.round(estBytes / 1048576), archivedRecords: done, archivedDays: rows, keepDays: keep }));
  Logger.log(log.getProperty('lastRun'));
}

/** Monday morning email. */
function weeklySummary() {
  var last = JSON.parse(PropertiesService.getScriptProperties().getProperty('lastRun') || '{}');
  MailApp.sendEmail(Session.getEffectiveUser().getEmail(), 'Lemonade Stand research data: weekly summary',
    'Records in Firebase now: ' + countRecords() + '\n' +
    'Estimated storage at last run: ' + (last.estMB || 0) + ' MB (free plan: 1024 MB)\n' +
    'Last night: archived ' + (last.archivedRecords || 0) + ' records (' + (last.archivedDays || 0) + ' game days), keeping ' + (last.keepDays || KEEP_DAYS) + ' days in Firebase.\n' +
    'Archive folder in Google Drive: ' + FOLDER + '\n');
}

/** Lets one phone create groups: run makeAdmin('<Device ID from the app's Settings>') once in the editor. */
function makeAdmin(uid) {
  if (!uid || !/^[A-Za-z0-9]{20,128}$/.test(uid)) throw new Error('Paste the Device ID from Settings in the app');
  api('patch', BASE + '/admins/' + uid, { fields: { t: { timestampValue: new Date().toISOString() } } });
  Logger.log('Admin added: ' + uid);
}
